# console:
#   Where the boot log is printed.
#     both:     HDMI display and serial
#     display:  HDMI display
#     serial:   Serial (the 4-pad UART header, 115200 8N1, 3.3V)
#   Note:
#     Due to https://github.com/systemd/systemd/issues/9899, only the last
#       console will be the primary console (/dev/console) which is the
#       only console the initramfs shell and the systemd log use.
#     So when set to "both", the serial console is made primary.
console="both"

# console="serial"

#   Use custom console arguments.
#   Note:
#     When this is set, the arguments generated by the "console" setting
#     above are ignored.
# consoleargs=""

# extraargs:
#   Extra kernel command line arguments.
#   The MXQ has only 512MB of RAM, so the image ships with zram swap
#   enabled. That is configured entirely in userspace by
#   /etc/default/armbian-zram-config -- there is no kernel command line
#   switch for it, so set ENABLED=false there (or mask
#   armbian-zram-config.service) if you want no swap. In particular
#   "zswap.enabled=0" does NOT turn it off: zswap is a different,
#   unrelated mechanism that this image does not use.
# extraargs=""


# fdtfile:
#   Which device tree the boot script loads from /dtb/.
#   Default here: meson8b-mxq-nand.dtb, the raw-NAND variant, because
#   probing the flash is what this image is for. It enables the NAND
#   controller and nothing else; the SD card slot is a different
#   controller and is unaffected, so if nand_scan() fails the box still
#   boots, just without any /dev/mtd.
#
#   The plain mainline device tree, with no NAND controller at all:
#
#     fdtfile=meson8b-mxq.dtb
#
#   Use it if the NAND one hangs the boot. The way back is to pull the
#   card and edit this file on another machine.
